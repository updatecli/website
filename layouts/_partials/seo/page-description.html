{{- /* A short description for pages whose content makes a poor one, or "" to keep the normal behaviour. Used by the seo/description, seo/opengraph and seo/twitter overrides, which otherwise fall back to the page summary. Also caps summary fallbacks at 160 characters for pages without a description. Changelog pages are generated by updateclibot and have no description, so the summary fallback put whole release notes (up to 14,000 characters) in the meta description. The line below is built from page data only (the project section, the title and the date), so it holds whatever the generator writes. */ -}} {{- $description := "" -}} {{- if eq .Section "changelogs" -}} {{- if .IsPage -}} {{- $project := "" -}} {{- with .Parent }}{{ with .File }}{{ $project = .ContentBaseName | humanize }}{{ end }}{{ end -}} {{- $description = printf "%s %s release notes, published %s." $project .Title (.Date.Format "2 January 2006") | strings.TrimSpace -}} {{- else -}} {{- $project := "" -}} {{- with .File }}{{ $project = .ContentBaseName | humanize }}{{ end -}} {{- $description = printf "Release notes for every %s release, newest first." $project | replaceRE `\s+` " " -}} {{- end -}} {{- else if not .Description -}} {{- /* Pages without a description (mostly blog posts) fall back to their summary, which can run to several hundred characters. Cut it at a word boundary so it fits the 160 characters search engines display. */ -}} {{- $summary := .Summary | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- if gt (len $summary) 160 -}} {{- $description = truncate 150 "..." $summary -}} {{- end -}} {{- end -}} {{- return $description -}}
